rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ---------- Identité / rôles --------------------------------------------
    function signedIn() { return request.auth != null; }

    // Admin = custom claim serveur `admin:true`  OU  UID explicite (immuable).
    // Plus aucun test sur l'email : un email peut être non vérifié ou réutilisé.
    function isAdmin() {
      return signedIn() && (
        request.auth.token.get('admin', false) == true
        || request.auth.uid in ['uid-marc-admin']
      );
    }

    // ---------- Liste BLANCHE des champs de /members ------------------------
    // Profil saisi par la personne
    function profileKeys() {
      return ['prenom','nom','email','whatsapp','pays','ville','age','service',
              'concepts_connus','tan_trading','objectif_inter','tan_disponib',
              'objectif_debutant','marches','platforme_siyal','niveau_risk',
              'engagement1','engagement2','engagement3'];
    }
    // Déclarations de paiement (informatives : l'admin vérifie, jamais un droit d'accès)
    function claimKeys() {
      return ['service','plan','plan_label_fr','plan_label_ht','plan_amount',
              'payment_method','txn_id','txn_ref'];
    }
    // Champs système que le client ne peut poser que sous conditions strictes
    function systemKeys() { return ['accountType','status','createdAt','updatedAt']; }
    function allowedKeys() { return profileKeys().concat(claimKeys()).concat(systemKeys()); }

    function incoming() { return request.resource.data; }

    function okStr(k, max) {
      return !(k in incoming()) || (incoming()[k] is string && incoming()[k].size() <= max);
    }
    function textOk() {
      return okStr('prenom',100) && okStr('nom',100) && okStr('email',254)
          && okStr('whatsapp',40) && okStr('pays',80) && okStr('ville',80)
          && okStr('objectif_inter',3000) && okStr('objectif_debutant',3000)
          && okStr('txn_id',120) && okStr('txn_ref',120);
    }
    // L'email du profil doit être celui du compte connecté (anti-usurpation)
    function emailMatches() {
      return !('email' in incoming())
          || incoming().email.lower() == request.auth.token.email.lower();
    }

    // status : jamais modifiable par le membre ; seulement posé à PENDING s'il n'existe pas
    function statusUnchangedOrInitPending() {
      return incoming().get('status', null) == resource.data.get('status', null)
          || (!('status' in resource.data) && incoming().status == 'PENDING');
    }
    // accountType : free -> applicant autorisé, rien d'autre
    function accountTypeSafe() {
      return incoming().get('accountType', null) == resource.data.get('accountType', null)
          || (resource.data.get('accountType','free') == 'free' && incoming().accountType == 'applicant');
    }
    // Déclarations de paiement verrouillées dès que le statut quitte PENDING
    function claimEditable() {
      return resource.data.get('status','PENDING') == 'PENDING'
          || !incoming().diff(resource.data).affectedKeys().hasAny(claimKeys());
    }
    function validStatus(s) { return s in ['PENDING','APPROVED','PAID','ADMITTED','REJECTED']; }

    match /u0/{userId} { allow update: if signedIn(); }
    match /u1/{userId} { allow update: if request.auth.uid == userId; }
    match /u2/{userId} { allow update: if incoming().diff(resource.data).affectedKeys().hasOnly(allowedKeys()); }
    match /u3/{userId} { allow update: if statusUnchangedOrInitPending(); }
    match /u4/{userId} { allow update: if incoming().get('status', null) == resource.data.get('status', null); }
    match /u5/{userId} { allow update: if (!('status' in resource.data) && incoming().status == 'PENDING'); }
    match /u6/{userId} { allow update: if accountTypeSafe(); }
    match /u7/{userId} { allow update: if claimEditable(); }
    match /u8/{userId} { allow update: if emailMatches(); }
    match /u9/{userId} { allow update: if textOk(); }
    match /u10/{userId} { allow update: if (incoming().get('createdAt',null) == resource.data.get('createdAt',null) || (!('createdAt' in resource.data) && incoming().createdAt == request.time)); }
    match /u11/{userId} { allow update: if (incoming().get('updatedAt',null) == resource.data.get('updatedAt',null) || incoming().updatedAt == request.time); }
    match /c0/{userId} { allow create: if signedIn() && request.auth.uid == userId && incoming().keys().hasOnly(allowedKeys()); }
    match /c1/{userId} { allow create: if signedIn() && request.auth.uid == userId && incoming().get('accountType','free') in ['free','applicant']; }
    match /c2/{userId} { allow create: if signedIn() && request.auth.uid == userId && (!('status' in incoming()) || incoming().status == 'PENDING'); }
    match /c3/{userId} { allow create: if signedIn() && request.auth.uid == userId && emailMatches(); }
    match /c4/{userId} { allow create: if signedIn() && request.auth.uid == userId && textOk(); }
    match /c5/{userId} { allow create: if signedIn() && request.auth.uid == userId && (!('createdAt' in incoming()) || incoming().createdAt == request.time); }
    match /c6/{userId} { allow create: if signedIn() && request.auth.uid == userId && (!('updatedAt' in incoming()) || incoming().updatedAt == request.time); }
  }
}
